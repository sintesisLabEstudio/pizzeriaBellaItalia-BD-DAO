          IdPedido (PK), IdCliente (FK), Cliente, Telefono,
          TamanoNombre, PrecioBaseTamano, MasaNombre, RecargoMasa,

          IdCliente (PK), Nombre, Telefono.
        CalculadoraPedido.cs  (clase static, sin dependencia de WinForms)
          Tamanos : Personal 5000 / Mediana 8500 (defecto) / Familiar 12000
          Masas   : Tradicional 0 (defecto) / Delgada 0 / Borde de Queso 1500
          DatosPedido: clase anidada con los valores de entrada de la interfaz.
          Calcular(DatosPedido) -> PedidoPizza:
            tamano y masa se normalizan al valor por defecto si no existen en el diccionario
            extras    = 800 * cantidad de ingredientes activos
            subtotal  = precioBase + recargoMasa + extras
            descuento = 10% del subtotal si AplicaConvenioEstudiante
            delivery  = 2000 si EsDelivery
            neto      = subtotal - descuento + delivery
            iva       = 19% del neto
            total     = neto + iva
        ConexionBD.cs  (helper estatico)
          CadenaConexion  : propiedad de solo lectura con la cadena indicada arriba.
          ObtenerConexion(): crea, abre y devuelve SqlConnection; si falla, hace Dispose y relanza.
          ProbarConexion(): bool; captura SqlException y la escribe en consola.

        4. CAPA DE ACCESO A DATOS (ADO.NET / DAO)
        -------------------------------------------------------------------------------
        PedidoDAO.cs  (6 metodos publicos + 3 privados)
          InsertarPedido(PedidoPizza) : bool
              BEGIN TRANSACTION -> ResolverIdCliente -> INSERT con 21 parametros
              -> SCOPE_IDENTITY -> COMMIT; si falla, ROLLBACK tolerante.
          ObtenerPedidosDataTable() : DataTable
              SELECT con INNER JOIN Clientes, ORDER BY IdPedido DESC.
          BuscarPedidosPorCliente(string) : DataTable
              WHERE c.Cliente LIKE '%' + filtro + '%'.
          ObtenerPedidoPorId(int) : PedidoPizza?
              SELECT completo con JOIN; devuelve null si no existe.
          ActualizarPedido(PedidoPizza) : bool
              ResolverIdCliente + UPDATE por IdPedido, todo en una transaccion.
          EliminarPedido(int) : bool
              DELETE por IdPedido.
          Privados: ResolverIdCliente (delega en ClienteDAO), Revertir (rollback seguro),
                    CargarParametros (carga los 21 parametros del INSERT/UPDATE).
        ClienteDAO.cs  (12 metodos publicos)
          EsClienteDuplicado(SqlException) : bool estatico -> ex.Number 2601 o 2627.
          ObtenerClientesDataTable() : DataTable -> ORDER BY IdCliente DESC.
          BuscarPorNombreYTelefono(n, t) : Cliente? -> control de duplicados en la UI.
          InsertarCliente(Cliente) : int -> INSERT + SCOPE_IDENTITY.
          InsertarClienteConTransaccion(c, conexion, tran) : int -> reutiliza la transaccion del llamador.
          ObtenerOCrearIdCliente(n, t, conexion, tran) : int -> busca o crea en la misma transaccion.
          ActualizarCliente(Cliente) : bool -> transaccion que actualiza Clientes y la copia
              de nombre/telefono guardada en Pedidos, para evitar datos divergentes.
          EliminarCliente(int) : bool -> DELETE por IdCliente.
          ValidaClienteSinPedidos(int) : bool -> COUNT(*) en Pedidos antes de borrar.
          BuscarClientesDataTable(criterio) : DataTable -> LIKE en Cliente o Telefono.
          ObtenerPedidosPorCliente(int) : DataTable -> utilitario de consulta.
          ObtenerClientePorId(int) : Cliente? -> utilitario de consulta.
        Notas de diseno del DAO:
          - Todas las consultas son parametrizadas (AddWithValue), sin concatenacion SQL.
          - Las operaciones que pueden dejar datos a medias usan SqlTransaction.
          - El rollback se hace en un helper que tolera transacciones ya finalizadas.

        7. FLUJOS PRINCIPALES
        -------------------------------------------------------------------------------
        Alta de pedido:
          Usuario completa cliente y opciones -> Calculo_Changed actualiza montos ->
          BtnCalcular_Click valida y confirma -> PedidoDAO.InsertarPedido abre transaccion ->
          ClienteDAO.ObtenerOCrearIdCliente reutiliza o crea cliente ->
          INSERT en Pedidos con SCOPE_IDENTITY -> COMMIT -> se limpia el formulario y se recarga la grilla.
        Actualizacion de pedido:
          El usuario selecciona una fila o busca por ID -> VistaPedido carga el detalle ->
          cambia datos -> btnActualizarPedido valida -> PedidoDAO.ActualizarPedido
          resuelve el cliente (puede crear uno nuevo) y hace UPDATE en la misma transaccion.
        Baja de cliente:
          FormClientes verifica ValidaClienteSinPedidos -> si tiene pedidos, se deniega ->
          si no, confirma y ejecuta DELETE.
        Seleccion de cliente desde el ABM:
          FormClientes.btnSelecciona -> FormPedido.CargarClienteDesdeSeleccion ->
          se cargan txtCliente y txtTelefono en el pedido.
        -------------------------------------------------------------------------------
        8. VALIDACION Y MANEJO DE ERRORES
        -------------------------------------------------------------------------------
        - Validaciones de interfaz: nombre y telefono obligatorios; ID positivo con int.TryParse;
          confirmacion para grabar, actualizar, eliminar y salir.
        - Duplicados: control previo por (Nombre, Telefono) y captura de SqlException 2601/2627
          traducida a MessageBox comprensible.
        - Integridad: no se puede borrar un cliente con pedidos; FK y UQ en la base.
        - Excepciones: cada metodo de la capa de presentacion muestra MessageBox y escribe
          estado en lblAyuda; los DAO relanzan despues de hacer rollback.

            using System.Collections.Generic;
            namespace PizzeriaBellaItalia.Dominio;
            // Reglas de negocio y cálculo de pedidos (desacopladas de la UI).
                CLASE: CalculadoraPedido
            public static class CalculadoraPedido
            {
                private const string TamanoPorDefecto = "Mediana";
                private const string MasaPorDefecto = "Masa Tradicional";
                // Opciones de tamaño: nombre -> precio base (CLP).
                public static readonly IReadOnlyDictionary<string, int> Tamanos = new Dictionary<string, int>
                {
                    { "Personal", 5000 },
                    { "Mediana", 8500 },
                    { "Familiar", 12000 }
                };
                // Opciones de masa: nombre -> recargo (CLP).
                public static readonly IReadOnlyDictionary<string, int> Masas = new Dictionary<string, int>
                {
                    { "Masa Tradicional", 0 },
                    { "Masa Delgada", 0 },
                    { "Borde de Queso", 1500 }
                };
                // Datos de entrada para el cálculo (sin dependencia de controles de UI).
                //CLASE: DatosPedido
                public sealed class DatosPedido // sealed: indica que no se puede heredar
                {
                    public string Cliente { get; set; } = "";
                    public string Telefono { get; set; } = "";
                    public string Tamano { get; set; } = TamanoPorDefecto;
                    public string Masa { get; set; } = MasaPorDefecto;
                    public bool QuesoExtra { get; set; }
                    public bool Pepperoni { get; set; }
                    public bool Champinones { get; set; }
                    public bool Aceitunas { get; set; }
                    public bool Tocino { get; set; }
                    public bool EsDelivery { get; set; }
                    public bool AplicaConvenioEstudiante { get; set; }
                }
                // Aplica reglas y retorna un PedidoPizza con montos calculados.
                // METODO/FUNCION: public static PedidoPizza Calcular(DatosPedido d)
                public static PedidoPizza Calcular(DatosPedido d)
                {
                    // Si el tamaño o la masa no existen, se usa el valor por defecto y se guarda ese nombre,
                    // para que el nombre persistido siempre coincida con el precio aplicado.
                    string tamano = Tamanos.ContainsKey(d.Tamano) ? d.Tamano : TamanoPorDefecto;
                    string masa = Masas.ContainsKey(d.Masa) ? d.Masa : MasaPorDefecto;
                    int precioBase = Tamanos[tamano];
                    int recargoMasa = Masas[masa];
                    int extras = (d.QuesoExtra ? 1 : 0) + (d.Pepperoni ? 1 : 0) +
                                 (d.Champinones ? 1 : 0) + (d.Aceitunas ? 1 : 0) +
                                 (d.Tocino ? 1 : 0);
                    int costoIng = extras * 800;
                    int subtotal = precioBase + recargoMasa + costoIng;
                    int descuento = d.AplicaConvenioEstudiante ? (int)(subtotal * 0.10) : 0;
                    int delivery = d.EsDelivery ? 2000 : 0;
                    int neto = subtotal - descuento + delivery;
                    int iva = (int)(neto * 0.19);
                    int total = neto + iva;
                    return new PedidoPizza // retorna un objeto de tipo PedidoPizza con los datos calculados
                    {
                        Cliente = d.Cliente,
                        Telefono = d.Telefono,
                        TamanoNombre = tamano,
                        PrecioBaseTamano = precioBase,
                        MasaNombre = masa,
                        RecargoMasa = recargoMasa,
                        QuesoExtra = d.QuesoExtra,
                        Pepperoni = d.Pepperoni,
                        Champinones = d.Champinones,
                        Aceitunas = d.Aceitunas,
                        Tocino = d.Tocino,
                        EsDelivery = d.EsDelivery,
                        AplicaConvenioEstudiante = d.AplicaConvenioEstudiante,
                        CostoIngredientes = costoIng,
                        SubtotalPizza = subtotal,
                        MontoDescuento = descuento,
                        CostoDelivery = delivery,
                        NetoAfecto = neto,
                        MontoIva = iva,
                        TotalPagar = total
                    };
                }
            }

                            return false;
                        }

                    public DataTable ObtenerPedidosPorCliente(int idCliente)
                    {
                        DataTable dt = new DataTable();

                                         FROM dbo.Pedidos WHERE IdCliente = @IdCliente ORDER BY IdPedido DESC";

                    // Cliente por su IdCliente (utilitario de consulta/pruebas).
                    // METODO/FUNCION: public Cliente? ObtenerClientePorId(int idCliente)
                    public Cliente? ObtenerClientePorId(int idCliente)
                    {
                        string query = "SELECT IdCliente, Cliente AS Nombre, Telefono FROM dbo.Clientes WHERE IdCliente = @IdCliente";
                        using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                        using (SqlCommand cmd = new SqlCommand(query, conexion))
                        {
                            cmd.Parameters.AddWithValue("@IdCliente", idCliente);
                            using (SqlDataReader r = cmd.ExecuteReader())
                            {
                                return r.Read() ? LeerCliente(r) : null;
                            }
                        }
                    }

                    // Convierte la fila actual del reader (IdCliente, Nombre, Telefono) en un Cliente.
                    // METODO/FUNCION: private static Cliente LeerCliente(SqlDataReader r) => new Cliente
                    private static Cliente LeerCliente(SqlDataReader r) => new Cliente
                    {
                        IdCliente = Convert.ToInt32(r["IdCliente"]),
                        Nombre = r["Nombre"]?.ToString() ?? "",
                        Telefono = r["Telefono"]?.ToString() ?? ""
                    };
                    // METODO/FUNCION: private static void AgregarParametrosActualizacion(SqlCommand cmd, Cliente c)
                    private static void AgregarParametrosActualizacion(SqlCommand cmd, Cliente c)
                    {
                        cmd.Parameters.AddWithValue("@IdCliente", c.IdCliente);
                        cmd.Parameters.AddWithValue("@Nombre", c.Nombre);
                        cmd.Parameters.AddWithValue("@Telefono", c.Telefono);
                    }
                    // Rollback tolerante: si la transacción ya terminó o la conexión se perdió, no oculta el error original.
                    // METODO/FUNCION: private static void Revertir(SqlTransaction tran)
                    private static void Revertir(SqlTransaction tran)
                    {
                        try { tran.Rollback(); }
                        catch (Exception) { /* la transacción ya no está activa */ }
                    }
                }

                    private int idClienteSeleccionado = 0; // 0 = cliente nuevo; > 0 = cliente en edición.
                    public FormClientes()
                    {
                        InitializeComponent();

                    // METODO/FUNCION: private void btnGrabar_Click(object? sender, EventArgs e)
                    private void btnGrabar_Click(object? sender, EventArgs e)
                    {
                        if (string.IsNullOrWhiteSpace(txtNombre.Text) || string.IsNullOrWhiteSpace(txtTelefono.Text))
                        {
                            MessageBox.Show("Por favor, ingrese el nombre y el teléfono del cliente.", "Datos incompletos", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                            return;
                        }
                        try
                        {
                            Cliente cliente = new Cliente
                            {
                                IdCliente = idClienteSeleccionado,
                                Nombre = txtNombre.Text.Trim(),
                                Telefono = txtTelefono.Text.Trim()
                            };
                            Cliente? duplicado = clienteDao.BuscarPorNombreYTelefono(cliente.Nombre, cliente.Telefono);
                        // -> FX: BuscarPorNombreYTelefono | ClienteDAO.cs | Busca un cliente por nombre y telefono para detectar duplicados.
                            if (duplicado != null && duplicado.IdCliente != idClienteSeleccionado)
                            {
                                MessageBox.Show(
                                    $"Ya existe un cliente con el nombre '{cliente.Nombre}' y el teléfono '{cliente.Telefono}'.\nUse otro teléfono o edite el cliente existente.",
                                    "Cliente duplicado", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                                txtTelefono.Focus();
                                return;
                            }
                            if (idClienteSeleccionado == 0)
                            {
                                int idGenerado = clienteDao.InsertarCliente(cliente);
                        // -> FX: InsertarCliente | ClienteDAO.cs | Da de alta un cliente y devuelve el IdCliente generado.
                                if (idGenerado > 0)
                                {
                                    MessageBox.Show("Cliente guardado con éxito.", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                                }
                            }
                            else
                            {
                                bool actualizado = clienteDao.ActualizarCliente(cliente);
                        // -> FX: ActualizarCliente | ClienteDAO.cs | Actualiza Clientes y la copia de datos en Pedidos dentro de una transaccion.
                                if (actualizado)
                                {
                                    MessageBox.Show("Cliente actualizado con éxito.", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                                }
                            }
                            PrepararNuevoCliente();
                            CargarClientes();
                        }
                        catch (SqlException ex) when (ClienteDAO.EsClienteDuplicado(ex))
                        // -> FX: EsClienteDuplicado | ClienteDAO.cs | Detecta SqlException 2601 o 2627 de la restriccion UNIQUE.
                        {
                            MessageBox.Show("Ya existe un cliente con ese nombre y teléfono. No se guardó el cambio.",
                                            "Cliente duplicado", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                        }
                        catch (Exception ex)
                        {
                            MessageBox.Show("Error al procesar el cliente:\n" + ex.Message, "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
                        }
                    }
                    // METODO/FUNCION: private void btnElimina_Click(object? sender, EventArgs e)
                    private void btnElimina_Click(object? sender, EventArgs e)
                    {
                        if (dgvClientes.CurrentRow == null || dgvClientes.CurrentRow.IsNewRow)
                        {
                            MessageBox.Show("Por favor, seleccione un cliente de la lista para eliminar.", "Atención", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                            return;
                        }
                        DataGridViewRow filaSeleccionada = dgvClientes.CurrentRow;
                        int idCliente = Convert.ToInt32(filaSeleccionada.Cells["IdCliente"].Value ?? 0);
                        string nombreCliente = filaSeleccionada.Cells["Nombre"].Value?.ToString() ?? "Sin nombre";
                        if (!clienteDao.ValidaClienteSinPedidos(idCliente))
                        // -> FX: ValidaClienteSinPedidos | ClienteDAO.cs | Verifica que el cliente no tenga pedidos antes de eliminarlo.
                        {
                            MessageBox.Show($"No es posible eliminar al cliente '{nombreCliente}' porque tiene pedidos registrados en el sistema.",
                                            "Acción Denegada", MessageBoxButtons.OK, MessageBoxIcon.Stop);
                            return;
                        }
