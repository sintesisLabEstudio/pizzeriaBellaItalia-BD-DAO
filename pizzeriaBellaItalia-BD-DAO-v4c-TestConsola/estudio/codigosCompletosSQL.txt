GLOSARIO RAPIDO DE T-SQL
    GO                       separa lotes de instrucciones; no es una instruccion SQL.
    USE nombre               cambia la base de datos activa.
    DB_ID / OBJECT_ID        consultan si existen una base o un objeto (tabla).
    CREATE DATABASE          crea una base de datos; FOR ATTACH la adjunta desde archivos.
    CREATE TABLE             define una tabla con columnas, tipos y restricciones.
    IDENTITY(1,1)            genera automaticamente un número correlativo.
    PRIMARY KEY / FOREIGN KEY claves primaria y foranea (integridad referencial).
    UNIQUE                   impide valores repetidos en una columna o combinacion.
    SET IDENTITY_INSERT      permite insertar valores explicitos en una columna IDENTITY.
    TRUNCATE TABLE           borra todas las filas y reinicia IDENTITY; no permite FK activa.
    DBCC CHECKIDENT RESEED   reinicia el contador de una columna IDENTITY.
    BACKUP DATABASE          genera un archivo .bak; WITH INIT lo sobrescribe.
    sp_detach_db             desadjunta una base conservando .mdf y .ldf (NO los borra).
    sys.databases            catalogo con estado y modo de acceso de las bases.
    sys.master_files         catalogo con la ruta fisica de los archivos de las bases.
    INFORMATION_SCHEMA       catalogo estandar de tablas, columnas y tipos.
    sp_executesql            ejecuta SQL dinamico y permite parametros OUTPUT.
    BEGIN TRY / CATCH        captura errores de ejecucion.
    SET XACT_ABORT ON        revierte toda la transaccion si una instruccion falla.

DESCRIPCION: Diagnostico de PizzeriaDB (mismo contenido que la version de comandos\BAT) disponible tambien en scriptsSql.
CODIGO:
    -- Diagnostico de PizzeriaDB: instancia, estado, modo de acceso, archivos y filas por tabla.
    -- Equivale a 03_verEstadoConexion-03.bat y a comandos\BAT\03_verEstadoConexion.sql.
    SET NOCOUNT ON;
    SELECT @@SERVERNAME AS Instancia, SERVERPROPERTY('ProductVersion') AS Version;
    DECLARE @estado NVARCHAR(60), @acceso NVARCHAR(60);
    SELECT @estado = state_desc, @acceso = user_access_desc
    FROM sys.databases
    WHERE name = N'PizzeriaDB';
    IF @estado IS NULL
    BEGIN
        PRINT 'RESULTADO: ERROR, la base PizzeriaDB no existe o esta desadjuntada. Ejecute 01 o 02.';
        RETURN;
    END
    SELECT name AS Base, state_desc AS Estado, user_access_desc AS Modo
    FROM sys.databases
    WHERE name = N'PizzeriaDB';
    SELECT name AS Archivo, type_desc AS Tipo, physical_name AS Ruta
    FROM sys.master_files
    WHERE database_id = DB_ID(N'PizzeriaDB');
    DECLARE @clientes INT, @pedidos INT;
    BEGIN TRY
        EXEC sys.sp_executesql
            N'SELECT @c = COUNT(*) FROM PizzeriaDB.dbo.Clientes; SELECT @p = COUNT(*) FROM PizzeriaDB.dbo.Pedidos;',
            N'@c INT OUTPUT, @p INT OUTPUT',
            @c = @clientes OUTPUT, @p = @pedidos OUTPUT;
        SELECT 'Clientes' AS Tabla, @clientes AS Filas
        UNION ALL
        SELECT 'Pedidos', @pedidos;
    END TRY
    BEGIN CATCH
        PRINT 'RESULTADO: ERROR, no se pudieron leer las tablas Clientes y Pedidos. Ejecute 01.';
        RETURN;
    END CATCH
    IF @estado = N'ONLINE' AND @acceso = N'MULTI_USER'
        PRINT 'RESULTADO: TODO OK, base ONLINE y MULTI_USER';
    ELSE
        PRINT 'RESULTADO: REVISAR, estado ' + @estado + ' y modo ' + @acceso;
    GO
PASO A PASO:
    1. SET NOCOUNT ON: salida limpia, sin contadores de filas.
    2. SELECT @@SERVERNAME y SERVERPROPERTY: identifica instancia y version.
    3. SELECT desde sys.databases: lee estado y modo de acceso de PizzeriaDB.
    4. IF @estado IS NULL: si no existe, imprime un error y sale.
    5. SELECT desde sys.master_files: muestra las rutas de los archivos fisicos.
    6. sp_executesql con OUTPUT y UNION ALL: cuenta filas de Clientes y Pedidos.
    7. BEGIN TRY / CATCH: controla errores de lectura de tablas.
    8. PRINT final con el resultado TODO OK o REVISAR.

ARCHIVO: 07_vaciaBaseDatos.sql
RUTA: C:\desarrollo\Csoft\pizzeriaBellaItalia-BD-DAO-v4B\Codigo_Funcional\BDpizzeria\scriptsSql\07_vaciaBaseDatos.sql
DESCRIPCION: Vacia las dos tablas y reinicia los IDENTITY dentro de una transaccion. Equivale a 06_vaciaBaseDatos-06.bat.
CODIGO:
    -- Vacia Clientes y Pedidos y reinicia los contadores IDENTITY.
    -- Equivale a 06_vaciaBaseDatos-06.bat (que ademas pide confirmacion con choice).
    -- Se elimina y se vuelve a crear la llave foranea porque TRUNCATE no puede
    -- ejecutarse mientras exista una FK que referencie la tabla.
    SET XACT_ABORT ON;
    BEGIN TRANSACTION;
    ALTER TABLE dbo.Pedidos DROP CONSTRAINT FK_Pedidos_Clientes;
    TRUNCATE TABLE dbo.Pedidos;
    TRUNCATE TABLE dbo.Clientes;
    DBCC CHECKIDENT ('dbo.Pedidos', RESEED, 0);
    DBCC CHECKIDENT ('dbo.Clientes', RESEED, 0);
    ALTER TABLE dbo.Pedidos ADD CONSTRAINT FK_Pedidos_Clientes
        FOREIGN KEY (IdCliente) REFERENCES dbo.Clientes (IdCliente);
    COMMIT TRANSACTION;
    GO
PASO A PASO:
    1. SET XACT_ABORT ON + BEGIN TRANSACTION: si cualquier instruccion falla, se revierte todo.
    2. ALTER TABLE ... DROP CONSTRAINT FK_Pedidos_Clientes: quita la FK porque TRUNCATE no puede ejecutarse con una referencia activa.
    3. TRUNCATE TABLE dbo.Pedidos y dbo.Clientes: elimina todas las filas (y es mas rapido que DELETE).
    4. DBCC CHECKIDENT (..., RESEED, 0): reinicia los contadores para que el proximo registro sea el número 1.
    5. ALTER TABLE ... ADD CONSTRAINT: restaura la llave foranea y la integridad referencial.
    6. COMMIT TRANSACTION: confirma el conjunto completo de cambios.

RUTA: C:\desarrollo\Csoft\pizzeriaBellaItalia-BD-DAO-v4B\Codigo_Funcional\BDpizzeria\scriptsSql\10_verRegistrosTablas.sql
DESCRIPCION: Visor de las copias de seguridad: adjunta temporalmente una copia de .mdf/.ldf, muestra tablas y registros, y la desadjunta. Lo invoca verRegistrosTablas.bat con -v VisorDir y ConLog.
CODIGO:
    -- ============================================================================
    -- VISOR DE LAS COPIAS DE PIZZERIADB - PIZZERIA BELLA ITALIA
    -- ----------------------------------------------------------------------------
    -- Adjunta como BD TEMPORARIA (PizzeriaDB_Visor) una copia de los archivos
    -- pizzeriaDB.mdf/.ldf que estan en esta carpeta respaldo, muestra tablas,
    -- estructura y registros, y al final DESADJUNTA esa BD temporal.
    --
    -- IMPORTANTE: se usa sp_detach_db y NO DROP DATABASE, porque DROP borraria
    -- los archivos .mdf/.ldf de la carpeta respaldo (que son el respaldo).
    --
    -- Lo invoca verRegistrosTablas.bat con:
    --   -v VisorDir="ruta\a\_visor" ConLog=1|0
    -- ============================================================================
    SET NOCOUNT ON;
    DECLARE @VisorDir NVARCHAR(500) = N'$(VisorDir)';
    DECLARE @ConLog   NVARCHAR(5)   = N'$(ConLog)';
    DECLARE @Mdf      NVARCHAR(500) = @VisorDir + N'\visor.mdf';
    DECLARE @Ldf      NVARCHAR(500) = @VisorDir + N'\visor.ldf';
    DECLARE @Sql      NVARCHAR(MAX);
    IF DB_ID(N'PizzeriaDB_Visor') IS NOT NULL
    BEGIN
        ALTER DATABASE [PizzeriaDB_Visor] SET SINGLE_USER WITH ROLLBACK IMMEDIATE;
        EXEC master.dbo.sp_detach_db @dbname = N'PizzeriaDB_Visor';
    END
    IF @ConLog = N'1'
        SET @Sql = N'CREATE DATABASE [PizzeriaDB_Visor] ON (FILENAME = N''' + @Mdf + N'''), (FILENAME = N''' + @Ldf + N''') FOR ATTACH;'
    ELSE
        SET @Sql = N'CREATE DATABASE [PizzeriaDB_Visor] ON (FILENAME = N''' + @Mdf + N''') FOR ATTACH_REBUILD_LOG;';
    BEGIN TRY
        EXEC sp_executesql @Sql;
    END TRY
    BEGIN CATCH
        RAISERROR('No se pudieron adjuntar las copias .mdf/.ldf de la carpeta respaldo.', 16, 1);
    END CATCH
    PRINT '';
    PRINT '=== 1) TABLAS Y CANTIDAD DE FILAS (DE LAS COPIAS EN BD) ===';
    SELECT t.name AS Tabla, SUM(p.rows) AS Filas
    FROM PizzeriaDB_Visor.sys.tables t
    LEFT JOIN PizzeriaDB_Visor.sys.partitions p
           ON p.object_id = t.object_id AND p.index_id IN (0,1)
    WHERE t.is_ms_shipped = 0
    GROUP BY t.name
    ORDER BY t.name;
    IF OBJECT_ID(N'PizzeriaDB_Visor.dbo.Clientes', N'U') IS NOT NULL
    BEGIN
        PRINT '';
        PRINT '=== 2) ESTRUCTURA DE Clientes ===';
        SELECT c.column_id AS Orden, c.name AS Columna, ty.name AS Tipo,
               c.max_length AS Largo, c.is_nullable AS AdmiteNull
        FROM PizzeriaDB_Visor.sys.columns c
        JOIN PizzeriaDB_Visor.sys.types ty ON ty.user_type_id = c.user_type_id
        WHERE c.object_id = OBJECT_ID(N'PizzeriaDB_Visor.dbo.Clientes')
        ORDER BY c.column_id;
        PRINT '';
        PRINT '=== 3) REGISTROS DE Clientes ===';
        SELECT * FROM [PizzeriaDB_Visor].dbo.Clientes;
    END
    IF OBJECT_ID(N'PizzeriaDB_Visor.dbo.Pedidos', N'U') IS NOT NULL
    BEGIN
        PRINT '';
        PRINT '=== 4) ESTRUCTURA DE Pedidos ===';
