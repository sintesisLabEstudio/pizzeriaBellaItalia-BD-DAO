                    //OBTIENE CONEXION ƒ
                    /// Abre y retorna una conexión activa a SQL Server Express.
                    public static SqlConnection ObtenerConexion()
                    {
                            SqlConnection conexion = new SqlConnection(cadenaConexion);
                            conexion.Open();
                            return conexion;
                    }
                    
                    //PROBAR CONEXION ƒ
                    /// Prueba el estado del servicio de SQL Express e informa la disponibilidad.
                    public static bool ProbarConexion()
                    {
                        try
                        {
                            using (SqlConnection conexion = ObtenerConexion())
                            {
                                return conexion.State == System.Data.ConnectionState.Open;
                            }
                        }
                        catch (SqlException ex)
                        {
                            Console.WriteLine($"[Error de Conexión SQL]: {ex.Message}");
                            return false;
                        }
                    }
                }
            }

    Archivo PedidoDAO.cs:
            using System;
            using System.Collections.Generic;
            using System.Data;
            using Microsoft.Data.SqlClient;
            using PizzeriaBellaItalia.Dominio;

            namespace PizzeriaBellaItalia.AccesoDatos
            {
                public class PedidoDAO
                {
                    //INSERTA PEDIDO ƒ
                    /// CREATE: Inserta un nuevo pedido de pizza en SQL Server Express.
                    public bool InsertarPedido(PedidoPizza pedido)
                    {
                        string query = @"INSERT INTO Pedidos (Cliente, Telefono, TamanoNombre, TotalPagar)
                                        VALUES (@Cliente, @Telefono, @TamanoNombre, @TotalPagar)";

                        using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                        using (SqlCommand cmd = new SqlCommand(query, conexion))
                        {
                            cmd.Parameters.AddWithValue("@Cliente", pedido.Cliente);
                            cmd.Parameters.AddWithValue("@Telefono", pedido.Telefono);
                            cmd.Parameters.AddWithValue("@TamanoNombre", pedido.TamanoNombre);
                            cmd.Parameters.AddWithValue("@TotalPagar", pedido.TotalPagar);

                            return cmd.ExecuteNonQuery() > 0;
                        }
                    }

                    //OBTIENE PEDIDOS ƒ
                    /// READ: Obtiene todos los pedidos registrados en un DataTable (Modelo Desconectado).
                    public DataTable ObtenerPedidosDataTable()
                    {
                        DataTable dt = new DataTable();
                        string query = "SELECT Id, Cliente, Telefono, TamanoNombre AS Tamaño, TotalPagar AS TotalPagar, FechaRegistro FROM Pedidos ORDER BY Id DESC";
                        
                        using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                        using (SqlDataAdapter adapter = new SqlDataAdapter(query, conexion))
                        {
                            adapter.Fill(dt);
                        }
                        return dt;
                    }

                    //BUSCA PEDIDO nombre cliente ƒ
                    /// READ: Busca pedidos filtrando por el nombre del cliente de forma dinámica (LIKE).
                    public DataTable BuscarPedidosPorCliente(string filtroCliente)
                    {
                        DataTable dt = new DataTable();
                        string query = @"SELECT Id, Cliente, Telefono, TamanoNombre AS Tamaño, TotalPagar AS Total, FechaRegistro
                                        FROM Pedidos
                                        WHERE Cliente LIKE @Filtro
                                        ORDER BY Id DESC";
                        using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                        using (SqlCommand cmd = new SqlCommand(query, conexion))
                        {
                            cmd.Parameters.AddWithValue("@Filtro", "%" + filtroCliente + "%");
                            using (SqlDataAdapter adapter = new SqlDataAdapter(cmd))
                            {
                                adapter.Fill(dt);
                            }
                        }
                        return dt;
                    }

                    //OBTIENE PEDIDO X id ƒ
                    /// READ: Obtiene un pedido específico por su ID primario.
                    public PedidoPizza? ObtenerPedidoPorId(int id)
                    {
                        string query = "SELECT Id, Cliente, Telefono, TamanoNombre, TotalPagar FROM Pedidos WHERE Id = @Id";
                        using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                        using (SqlCommand cmd = new SqlCommand(query, conexion))
                        {
                            cmd.Parameters.AddWithValue("@Id", id);
                            using (SqlDataReader reader = cmd.ExecuteReader())
                            {
                                if (reader.Read())
                                {
                                    return new PedidoPizza
                                    {
                                        Id = Convert.ToInt32(reader["Id"]),
                                        Cliente = reader["Cliente"].ToString() ?? "",
                                        Telefono = reader["Telefono"].ToString() ?? "",
                                        TamanoNombre = reader["TamanoNombre"].ToString() ?? "",
                                        TotalPagar = Convert.ToInt32(reader["TotalPagar"])
                                    };
                                }
                            }
                        }
                        return null;
                    }

                    //ACTUALIZA PEDIDO existente  ƒ      
                    /// UPDATE: Modifica los datos de un pedido existente.
                    public bool ActualizarPedido(PedidoPizza pedido)
                    {
                        string query = @"UPDATE Pedidos
                                        SET Cliente = @Cliente,
                                            Telefono = @Telefono,
                                            TamanoNombre = @TamanoNombre,
                                            TotalPagar = @TotalPagar
                                        WHERE Id = @Id";
                        using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                        using (SqlCommand cmd = new SqlCommand(query, conexion))
                        {
                            cmd.Parameters.AddWithValue("@Id", pedido.Id);
                            cmd.Parameters.AddWithValue("@Cliente", pedido.Cliente);
                            cmd.Parameters.AddWithValue("@Telefono", pedido.Telefono);
                            cmd.Parameters.AddWithValue("@TamanoNombre", pedido.TamanoNombre);
                            cmd.Parameters.AddWithValue("@TotalPagar", pedido.TotalPagar);
                            return cmd.ExecuteNonQuery() > 0;
                        }
                    }

                    //ELIMINA PEDIDO x id ƒ
                    /// DELETE: Elimina un registro de pedido por su ID.
                    public bool EliminarPedido(int id)
                    {
                        string query = "DELETE FROM Pedidos WHERE Id = @Id";
                        using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                        using (SqlCommand cmd = new SqlCommand(query, conexion))
                        {
                            cmd.Parameters.AddWithValue("@Id", id);
                            return cmd.ExecuteNonQuery() > 0;
                        }
                    }
                }
            }

CREATE TABLE dbo.Pedidos
(
    IdPedido                  INT IDENTITY(1,1) NOT NULL
                                CONSTRAINT PK_Pedidos PRIMARY KEY,
    -- Datos del cliente
    Cliente                   NVARCHAR(100)     NOT NULL,
    Telefono                  NVARCHAR(30)      NOT NULL,
    -- Configuración de la pizza
    TamanoNombre              NVARCHAR(50)      NOT NULL,
    PrecioBaseTamano          INT               NOT NULL,
    MasaNombre                NVARCHAR(50)      NOT NULL,
    RecargoMasa               INT               NOT NULL,
    -- Ingredientes adicionales (flags)
    QuesoExtra                BIT               NOT NULL CONSTRAINT DF_Pedidos_QuesoExtra   DEFAULT (0),
    Pepperoni                 BIT               NOT NULL CONSTRAINT DF_Pedidos_Pepperoni    DEFAULT (0),
    Champinones               BIT               NOT NULL CONSTRAINT DF_Pedidos_Champinones  DEFAULT (0),
    Aceitunas                 BIT               NOT NULL CONSTRAINT DF_Pedidos_Aceitunas    DEFAULT (0),
    Tocino                    BIT               NOT NULL CONSTRAINT DF_Pedidos_Tocino       DEFAULT (0),
    -- Opciones de entrega y promoción
    EsDelivery                BIT               NOT NULL CONSTRAINT DF_Pedidos_EsDelivery   DEFAULT (0),
    AplicaConvenioEstudiante  BIT               NOT NULL CONSTRAINT DF_Pedidos_Convenio     DEFAULT (0),
    -- Montos calculados por el dominio (PedidoPizza)
    CostoIngredientes         INT               NOT NULL CONSTRAINT DF_Pedidos_CostoIng     DEFAULT (0),
    SubtotalPizza             INT               NOT NULL CONSTRAINT DF_Pedidos_Subtotal     DEFAULT (0),
    MontoDescuento            INT               NOT NULL CONSTRAINT DF_Pedidos_Descuento    DEFAULT (0),
    CostoDelivery             INT               NOT NULL CONSTRAINT DF_Pedidos_CostoDeliv   DEFAULT (0),
    NetoAfecto                INT               NOT NULL CONSTRAINT DF_Pedidos_Neto         DEFAULT (0),
    MontoIva                  INT               NOT NULL CONSTRAINT DF_Pedidos_Iva          DEFAULT (0),
    TotalPagar                INT               NOT NULL CONSTRAINT DF_Pedidos_Total        DEFAULT (0),
    -- Auditoría
    FechaCreacion             DATETIME          NOT NULL CONSTRAINT DF_Pedidos_FechaCreacion DEFAULT (GETDATE())
);
