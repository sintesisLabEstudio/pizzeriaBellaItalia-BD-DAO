    using System.Collections.Generic;
    namespace PizzeriaBellaItalia.Dominio;
    // Reglas de negocio y cálculo de pedidos (desacopladas de la UI).
        CLASE: CalculadoraPedido
    public static class CalculadoraPedido
    {
        private const string TamanoPorDefecto = "Mediana";
        private const string MasaPorDefecto = "Masa Tradicional";
        // Opciones de tamaño: nombre -> precio base (CLP).
        public static readonly IReadOnlyDictionary<string, int> Tamanos = new Dictionary<string, int>
        {
            { "Personal", 5000 },
            { "Mediana", 8500 },
            { "Familiar", 12000 }
        };
        // Opciones de masa: nombre -> recargo (CLP).
        public static readonly IReadOnlyDictionary<string, int> Masas = new Dictionary<string, int>
        {
            { "Masa Tradicional", 0 },
            { "Masa Delgada", 0 },
            { "Borde de Queso", 1500 }
        };
        // Datos de entrada para el cálculo (sin dependencia de controles de UI).
        //CLASE: DatosPedido
        public sealed class DatosPedido // sealed: indica que no se puede heredar
        {
            public string Cliente { get; set; } = "";
            public string Telefono { get; set; } = "";
            public string Tamano { get; set; } = TamanoPorDefecto;
            public string Masa { get; set; } = MasaPorDefecto;
            public bool QuesoExtra { get; set; }
            public bool Pepperoni { get; set; }
            public bool Champinones { get; set; }
            public bool Aceitunas { get; set; }
            public bool Tocino { get; set; }
            public bool EsDelivery { get; set; }
            public bool AplicaConvenioEstudiante { get; set; }
        }
        // Aplica reglas y retorna un PedidoPizza con montos calculados.
        // METODO/FUNCION: public static PedidoPizza Calcular(DatosPedido d)
        public static PedidoPizza Calcular(DatosPedido d)
        {
            // Si el tamaño o la masa no existen, se usa el valor por defecto y se guarda ese nombre,
            // para que el nombre persistido siempre coincida con el precio aplicado.
            string tamano = Tamanos.ContainsKey(d.Tamano) ? d.Tamano : TamanoPorDefecto;
            string masa = Masas.ContainsKey(d.Masa) ? d.Masa : MasaPorDefecto;
            int precioBase = Tamanos[tamano];
            int recargoMasa = Masas[masa];
            int extras = (d.QuesoExtra ? 1 : 0) + (d.Pepperoni ? 1 : 0) +
                         (d.Champinones ? 1 : 0) + (d.Aceitunas ? 1 : 0) +
                         (d.Tocino ? 1 : 0);
            int costoIng = extras * 800;
            int subtotal = precioBase + recargoMasa + costoIng;
            int descuento = d.AplicaConvenioEstudiante ? (int)(subtotal * 0.10) : 0;
            int delivery = d.EsDelivery ? 2000 : 0;
            int neto = subtotal - descuento + delivery;
            int iva = (int)(neto * 0.19);
            int total = neto + iva;
            return new PedidoPizza // retorna un objeto de tipo PedidoPizza con los datos calculados
            {
                Cliente = d.Cliente,
                Telefono = d.Telefono,
                TamanoNombre = tamano,
                PrecioBaseTamano = precioBase,
                MasaNombre = masa,
                RecargoMasa = recargoMasa,
                QuesoExtra = d.QuesoExtra,
                Pepperoni = d.Pepperoni,
                Champinones = d.Champinones,
                Aceitunas = d.Aceitunas,
                Tocino = d.Tocino,
                EsDelivery = d.EsDelivery,
                AplicaConvenioEstudiante = d.AplicaConvenioEstudiante,
                CostoIngredientes = costoIng,
                SubtotalPizza = subtotal,
                MontoDescuento = descuento,
                CostoDelivery = delivery,
                NetoAfecto = neto,
                MontoIva = iva,
                TotalPagar = total
            };
        }
    }

                    conexion.Open();
                    return conexion;

                catch

                    conexion.Dispose(); // Si no abre, no queda una conexión sin liberar.

            {
                try
                {
                    using (SqlConnection conexion = ObtenerConexion())
                        return conexion.State == ConnectionState.Open;

                    return false;

                {
                    if (pedidoDao.EliminarPedido(id))

                    {

            private void BtnCalcular_Click(object? sender, EventArgs e)
            {
                // Valida que el nombre y el teléfono del cliente estén ingresados.
                if (!ClienteYTelefonoValidos()) return;
                // Construye un objeto PedidoPizza con todos los datos y montos calculados desde la pantalla.
                var p = ConstruirPedido();
                // Muestra los montos en la sección "Detalle de Pago" (subtotal, descuento, delivery, IVA y total).
                MostrarResumen(p);
                // Si confirma, llama a InsertarPedido y refresca la grilla.
                if (MessageBox.Show($"¿Grabar pedido?\n\nTotal: ${p.TotalPagar:N0} CLP", "Confirmar", MessageBoxButtons.YesNo, MessageBoxIcon.Question) == DialogResult.Yes)
                {
                    try
                    {

                        {

                    catch (SqlException ex) when (ClienteDAO.EsClienteDuplicado(ex))
                    {
                        // Violación de UQ_Clientes_Cliente_Telefono al resolver/crear el cliente.
                        MostrarEstado("Cliente duplicado: nombre y teléfono ya registrados.", false);
                        MessageBox.Show("Ya existe un cliente con ese nombre y teléfono.\nSeleccione el cliente desde 'Clientes' o cambie los datos.",

                    }
                    catch (Exception ex)
                    {
                        MostrarEstado("Error al grabar el pedido.", false);

            {
                cargandoFormulario = true;   // evita recalculos mientras se llenan los combos

            private void LlenaForm(PedidoPizza p)
            {
                cargandoFormulario = true;   // evita recalculos mientras se asignan los controles
                txtIdPedido.Text = p.IdPedido.ToString();
                txtCliente.Text = p.Cliente; txtTelefono.Text = p.Telefono;
                cboTamano.SelectedItem = p.TamanoNombre; if (cboTamano.SelectedIndex < 0) cboTamano.Text = p.TamanoNombre;
                cboMasa.SelectedItem = p.MasaNombre; if (cboMasa.SelectedIndex < 0) cboMasa.Text = p.MasaNombre;
                chkQuesoExtra.Checked = p.QuesoExtra; chkPepperoni.Checked = p.Pepperoni; chkChampinones.Checked = p.Champinones;
                chkAceitunas.Checked = p.Aceitunas; chkTocino.Checked = p.Tocino;
                rdbDelivery.Checked = p.EsDelivery; rdbRetiro.Checked = !p.EsDelivery;
                chkConvenioEstudiante.Checked = p.AplicaConvenioEstudiante;
                cargandoFormulario = false;

            public FormClientes()

            // METODO/FUNCION: private void btnGrabar_Click(object? sender, EventArgs e)
            private void btnGrabar_Click(object? sender, EventArgs e)
            {
                if (string.IsNullOrWhiteSpace(txtNombre.Text) || string.IsNullOrWhiteSpace(txtTelefono.Text))
                {
                    MessageBox.Show("Por favor, ingrese el nombre y el teléfono del cliente.", "Datos incompletos", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return;
                }
                try
                {
                    Cliente cliente = new Cliente
                    {
                        IdCliente = idClienteSeleccionado,
                        Nombre = txtNombre.Text.Trim(),
                        Telefono = txtTelefono.Text.Trim()
                    };
                    Cliente? duplicado = clienteDao.BuscarPorNombreYTelefono(cliente.Nombre, cliente.Telefono);
                // -> FX: BuscarPorNombreYTelefono | ClienteDAO.cs | Busca un cliente por nombre y telefono para detectar duplicados.
                    if (duplicado != null && duplicado.IdCliente != idClienteSeleccionado)
                    {
                        MessageBox.Show(
                            $"Ya existe un cliente con el nombre '{cliente.Nombre}' y el teléfono '{cliente.Telefono}'.\nUse otro teléfono o edite el cliente existente.",
                            "Cliente duplicado", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                        txtTelefono.Focus();
                        return;
                    }
                    if (idClienteSeleccionado == 0)
                    {
                        int idGenerado = clienteDao.InsertarCliente(cliente);
                // -> FX: InsertarCliente | ClienteDAO.cs | Da de alta un cliente y devuelve el IdCliente generado.
                        if (idGenerado > 0)
                        {
                            MessageBox.Show("Cliente guardado con éxito.", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                        }
                    }
                    else
                    {
                        bool actualizado = clienteDao.ActualizarCliente(cliente);
                // -> FX: ActualizarCliente | ClienteDAO.cs | Actualiza Clientes y la copia de datos en Pedidos dentro de una transaccion.
                        if (actualizado)
                        {
                            MessageBox.Show("Cliente actualizado con éxito.", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                        }
                    }
                    PrepararNuevoCliente();
                    CargarClientes();
                }
                catch (SqlException ex) when (ClienteDAO.EsClienteDuplicado(ex))
                // -> FX: EsClienteDuplicado | ClienteDAO.cs | Detecta SqlException 2601 o 2627 de la restriccion UNIQUE.
                {
                    MessageBox.Show("Ya existe un cliente con ese nombre y teléfono. No se guardó el cambio.",
                                    "Cliente duplicado", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                }
                catch (Exception ex)
                {
                    MessageBox.Show("Error al procesar el cliente:\n" + ex.Message, "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
            // METODO/FUNCION: private void btnElimina_Click(object? sender, EventArgs e)
            private void btnElimina_Click(object? sender, EventArgs e)
            {
                if (dgvClientes.CurrentRow == null || dgvClientes.CurrentRow.IsNewRow)
                {
                    MessageBox.Show("Por favor, seleccione un cliente de la lista para eliminar.", "Atención", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return;
                }
                DataGridViewRow filaSeleccionada = dgvClientes.CurrentRow;
                int idCliente = Convert.ToInt32(filaSeleccionada.Cells["IdCliente"].Value ?? 0);
                string nombreCliente = filaSeleccionada.Cells["Nombre"].Value?.ToString() ?? "Sin nombre";
                if (!clienteDao.ValidaClienteSinPedidos(idCliente))
                // -> FX: ValidaClienteSinPedidos | ClienteDAO.cs | Verifica que el cliente no tenga pedidos antes de eliminarlo.
                {
                    MessageBox.Show($"No es posible eliminar al cliente '{nombreCliente}' porque tiene pedidos registrados en el sistema.",
                                    "Acción Denegada", MessageBoxButtons.OK, MessageBoxIcon.Stop);
                    return;
                }
                DialogResult respuesta = MessageBox.Show($"¿Está seguro que desea eliminar al cliente '{nombreCliente}'?",
                                                         "Confirmar Eliminación",
                                                         MessageBoxButtons.YesNo,
                                                         MessageBoxIcon.Question);
                if (respuesta == DialogResult.Yes)
                {
                    try
                    {
                        if (clienteDao.EliminarCliente(idCliente))
                // -> FX: EliminarCliente | ClienteDAO.cs | Da de baja un cliente por su IdCliente.
                        {
                            MessageBox.Show("Cliente eliminado correctamente.", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                            PrepararNuevoCliente();
                            CargarClientes();
                        }
                    }
                    catch (Exception ex)
                    {
                        MessageBox.Show("Error al intentar eliminar:\n" + ex.Message, "Error de BD", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    }
                }
            }

            private void btnSelecciona_Click(object? sender, EventArgs e)
            {
                // Envía el cliente indicado en los cuadros de texto al formulario de pedido.
                if (string.IsNullOrWhiteSpace(txtNombre.Text) || string.IsNullOrWhiteSpace(txtTelefono.Text))
                {

                string nombre = txtNombre.Text.Trim();
                string telefono = txtTelefono.Text.Trim();
                FormPedido? frmPedido = Application.OpenForms.OfType<FormPedido>().FirstOrDefault();
                if (frmPedido != null)
                {
                    frmPedido.CargarClienteDesdeSeleccion(nombre, telefono);
                // -> FX: CargarClienteDesdeSeleccion | FormPedido.cs | Recibe el cliente elegido en FormClientes y carga los cuadros de texto.
                    frmPedido.BringToFront();

                }

                    nuevoPedido.Show();
                    nuevoPedido.CargarClienteDesdeSeleccion(nombre, telefono);
                    if (Application.OpenForms[0] == this)
                    {
                        nuevoPedido.FormClosed += (s, args) => this.Close();

            // METODO/FUNCION: private void CargarClientes(string criterio = "")
            private void CargarClientes(string criterio = "")
            {
                try
                {
                    if (string.IsNullOrWhiteSpace(criterio))
                        dgvClientes.DataSource = clienteDao.ObtenerClientesDataTable();
                // -> FX: ObtenerClientesDataTable | ClienteDAO.cs | Devuelve todos los clientes para la grilla.
                    else
                        dgvClientes.DataSource = clienteDao.BuscarClientesDataTable(criterio);
                // -> FX: BuscarClientesDataTable | ClienteDAO.cs | Filtra clientes por nombre o telefono con LIKE.
                    dgvClientes.AllowUserToAddRows = false;
                    dgvClientes.ReadOnly = true;
                    dgvClientes.SelectionMode = DataGridViewSelectionMode.FullRowSelect;
                    var colNombre = dgvClientes.Columns["Nombre"];
                    if (colNombre != null) colNombre.HeaderText = "Nombre Cliente";
                    var colTelefono = dgvClientes.Columns["Telefono"];
                    if (colTelefono != null) colTelefono.HeaderText = "Teléfono actual";
                }
                catch (Exception ex)
                {
                    MessageBox.Show("No se pudieron cargar los clientes.\n" + ex.Message, "Base de datos", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                }
            }

            private void PrepararNuevoCliente()
            {
                txtNombre.Clear();

            private void PresentaCamposParaEdicion()
            {
                // Pasa la fila seleccionada a los cuadros de texto y deja el formulario en modo "Actualizar".
                if (dgvClientes.CurrentRow == null || dgvClientes.CurrentRow.IsNewRow) return;
                DataGridViewRow fila = dgvClientes.CurrentRow;

                txtNombre.Text = fila.Cells["Nombre"].Value?.ToString() ?? "";
                txtTelefono.Text = fila.Cells["Telefono"].Value?.ToString() ?? "";
                btnGrabar.Text = "Actualizar";
                txtNombre.Focus();
            }
