A) btnBuscaPedidosXCliente_Click  (lee txtCliente)
  |
  +-- filtro vacio -> lblAyuda roja + MessageBox validacion + Focus -> FIN

        |   WHERE c.Cliente LIKE @Filtro
        v

        +-- oculta IdPedido / IdCliente
        +-- titulos visibles
        +-- lblAyuda verde "Busqueda por cliente realizada."

B) btnBuscarPedidoPorId_Click  (lee txtIdPedido)
  |
  +-- int.TryParse <= 0 -> MessageBox "Ingrese un ID valido" + Focus -> FIN
  +-- ID valido ->
        |
        v
  PedidoDAO.ObtenerPedidoPorId(id) -> PedidoPizza o null
        |
        +-- DataTable temporal:
        |     IdPedido | Cliente | Telefono | TamanoNombre | TotalPagar
        |     (1 fila si existe, vacia si no)
        |     -> dgvPedidos.DataSource + titulos
        |
        +-- null -> lblAyuda roja + MessageBox "No se encontro el pedido"
        |
        +-- existe -> lblAyuda verde
              |
              v
        vistaPedido(id)
              |
              +-- PedidoDAO.ObtenerPedidoPorId(id) -> PedidoPizza
              +-- LlenaForm(p):
              |     txtIdPedido, txtCliente, txtTelefono
              |     cboTamano, cboMasa
              |     5 chk* + rdbDelivery/rdbRetiro + chkConvenio
              +-- MostrarResumen(p) -> 5 labels

C) btnTodosLosPedidos_Click -> CargarPedidos() -> dgvPedidos completa

D) dgvPedidos_CellClick (columna oculta IdPedido)

CLIENTES  ||---o{  PEDIDOS
          relacion: 1:N por IdCliente
          FK: dbo.Pedidos.IdCliente -> dbo.Clientes.IdCliente
          ON UPDATE CASCADE / ON DELETE NO ACTION

CLIENTES
  IdCliente   INT IDENTITY  PK
  Cliente     NVARCHAR(100) UK   \
  Telefono    NVARCHAR(30)  UK   /  UQ_Clientes_Cliente_Telefono (UNIQUE)
                                     + rechaza SqlException 2601/2627

PEDIDOS  (21 columnas + auditoria)
  IdPedido                 INT IDENTITY PK
  IdCliente                INT FK
  Cliente, Telefono        NVARCHAR
  TamanoNombre, PrecioBaseTamano
  MasaNombre, RecargoMasa
  QuesoExtra, Pepperoni, Champinones, Aceitunas, Tocino   BIT
  EsDelivery, AplicaConvenioEstudiante                     BIT
  CostoIngredientes, SubtotalPizza, MontoDescuento
  CostoDelivery, NetoAfecto, MontoIva, TotalPagar          INT
  FechaCreacion            DATETIME DEFAULT GETDATE()


================================================================================
11. RESUMEN DEL VIAJE DE LOS DATOS
================================================================================

1) Entrada del usuario
   -> controles WinForms declarados en *.Designer.cs

2) Evento
   -> metodo *_Click / *_Changed / CellClick en FormPedido.cs o FormClientes.cs

3) Transformacion
   -> DatosPedido -> CalculadoraPedido.Calcular() -> PedidoPizza (objeto con montos)

4) Persistencia
   -> PedidoDAO / ClienteDAO -> ConexionBD.ObtenerConexion()
   -> SQL parametrizado -> PizzeriaDB (LocalDB)

5) Respuesta en pantalla
   -> DataTable .............. dgvPedidos / dgvClientes
   -> PedidoPizza ............ LlenaForm + MostrarResumen (5 labels)
   -> bool / SqlException .... MessageBox (incluido "Cliente duplicado")
   -> string ................. lblAyuda con color verde / rojo / blanco
